Engineering principles

Code examples

Các đoạn mã ngắn, minh họa tư duy kỹ thuật. Không chứa khóa API hay bí mật; hãy lấy cấu hình từ biến môi trường.

validate.js
// Kiểm tra dữ liệu đầu vào trước khi xử lý
export function parseEmail(input) {
  const value = String(input ?? "").trim();
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)) {
    throw new Error("Email không hợp lệ");
  }
  return value.toLowerCase();
}
client.js
// Gọi API với timeout và xử lý lỗi rõ ràng
export async function getJson(url, ms = 5000) {
  const ctrl = new AbortController();
  const t = setTimeout(() => ctrl.abort(), ms);
  try {
    const res = await fetch(url, { signal: ctrl.signal });
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    return await res.json();
  } finally { clearTimeout(t); }
}
